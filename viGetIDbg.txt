$oDev('t')=modbusread 1i 0
if $oDev.Item('t')="" then
  # non-modbus device, try VISA/SCPI
  $oDev('t')=*IDN?
  if $oDev.Item('t')="" then
    eval __viToggleMode($oDev)
    $oDev('t')=*IDN?
  endif
  $oDev('viIdn')=eval $oDev.Item('t')
  
else
  #modbus device F4T

  #determine DataMap
  $oDev('t')=modbusread 1i 100
  $oDev('f4tDataMap')=eval ((Int($oDev.Item('t') / 0x100) = 0x83) Or ($oDev.Item('t') = 0)) ? 1 : 2

  # determine Word Order
  $oDev('f4tWordOrder')=eval 1
  if $oDev.Item('f4tDataMap')=1 then
    $oDev('t')=modbusread 1i 14092
    if BitOR($oDev.Item('t'),1)<>1331 then
      $oDev('t')=modbusread 1i 6736
    endif
    if BitOR($oDev.Item('t'),1)=1331 then
      $oDev('t')=eval Int(Not BitAND($oDev.Item('t'), 1))
    endif
    $oDev('f4tWordOrder')=eval $oDev.Item('t')
  endif

  #get temperature units
  if $oDev.Item('f4tDataMap')=1 then
    $oDev('t')=modbusread 1i 6730
    $oDev('t')=eval Int($oDev.Item('t'))=15
  else
    $oDev('t')=modbusread 1i 901
    $oDev('t')=eval Int(BitAND($oDev.Item('t'),1))
  endif
  $oDev('f4tUnits')=eval $oDev.Item('t')

  # build an ID string
  if $oDev.Item('f4tDataMap')=1 then
    $oDev('tSN')=modbusread 2i 12
    $oDev('tRel')=modbusread 2i 3
    $oDev('tMfg')=modbusread 30s 46
    $oDev('tMod')=modbusread 15s 16
    $oDev('viIdn')=eval StringStripWS($oDev.Item('tMfg'),3) & "," & StringStripWS($oDev.Item('tMod'),3) & "," & $oDev.Item('tSN') & "," & $oDev.Item('tRel')
    eval $oDev.Remove('tSN')
    eval $oDev.Remove('tRel')
    eval $oDev.Remove('tMfg')
    eval $oDev.Remove('tMod')
  else
    $oDev('tRe1')=modbusread 1i 4
    $oDev('tRe2')=modbusread 1i 30
    $oDev('tMfg')=modbusread 1i 0
    $oDev('tMod')=modbusread 1i 3
    $oDev('viIdn')=eval $oDev.Item('tMfg') & "," & $oDev.Item('tMod') & "," & StringFormat("%.2f",$oDev.Item('tRe2')/100) & "," & StringFormat("%02i.%05i", $oDev.Item('tRe1'), $oDev.Item('tRe2'))
    eval $oDev.Remove('tRe1')
    eval $oDev.Remove('tRe2')
    eval $oDev.Remove('tMfg')
    eval $oDev.Remove('tMod')
  endif
  
endif

eval $oDev.Remove('t')
# IDN should now be in $oDev.Item('viIdn')
